--!strict
-- FPS Utility Suite - Mobile Silent Aim & Aimbot Suite
local success, err = pcall(function()
	local Rayfield = loadstring(game:HttpGet('https://sirius.menu/rayfield'))()

	local Players = game:GetService("Players")
	local RunService = game:GetService("RunService")
	local Workspace = game:GetService("Workspace")
	local LocalPlayer = Players.LocalPlayer
	local Camera = Workspace.CurrentCamera

	-- Configuration State
	local Config = {
		Combat = {
			Aimbot = false,
			AimbotFOV = 120,
			Smoothing = 8,
			
			SilentAim = false,
			SilentAimFOV = 180,
			DrawSilentFOV = true,
			DrawAimbotFOV = true,
			
			TargetPart = "Head",
		},
		Visuals = {
			ESPEnabled = false,
			BoxESP = false,
			Tracers = false,
		},
		Misc = {
			SpeedMultiplier = 16,
			JumpPower = 50,
		}
	}

	---------------------------------------------------------
	-- WINDOW INITIALIZATION
	---------------------------------------------------------
	local Window = Rayfield:CreateWindow({
	   Name = "Project Aura | Mobile FPS Suite",
	   LoadingTitle = "FPS Utility Suite (Mobile)",
	   LoadingSubtitle = "Advanced Silent Aim & ESP",
	   ConfigurationSaving = {
	      Enabled = true,
	      FolderName = "ProjectAuraMobile",
	      FileName = "FPSConfig"
	   },
	   KeySystem = false,
	})

	---------------------------------------------------------
	-- TABS
	---------------------------------------------------------
	local CombatTab = Window:CreateTab("Combat", 4483362458)
	local AimTab = Window:CreateTab("Aim Settings", 4483362458)
	local VisualsTab = Window:CreateTab("Visuals", 4483362458)
	local MiscTab = Window:CreateTab("Miscellaneous", 4483362458)
	local SettingsTab = Window:CreateTab("Settings", 4483362458)

	---------------------------------------------------------
	-- 1. COMBAT TAB (Silent Aim & Aimbot)
	---------------------------------------------------------
	CombatTab:CreateSection("Silent Aim (Bullet Curving)")

	CombatTab:CreateToggle({
	   Name = "Silent Aim Enable",
	   CurrentValue = Config.Combat.SilentAim,
	   Flag = "SilentAim",
	   Callback = function(Value)
	      Config.Combat.SilentAim = Value
	   end,
	})

	CombatTab:CreateToggle({
	   Name = "Draw Silent Aim Circle",
	   CurrentValue = Config.Combat.DrawSilentFOV,
	   Flag = "DrawSilentFOV",
	   Callback = function(Value)
	      Config.Combat.DrawSilentFOV = Value
	   end,
	})

	CombatTab:CreateSlider({
	   Name = "Silent Aim Radius",
	   Range = {50, 400},
	   Increment = 5,
	   Suffix = "px",
	   CurrentValue = Config.Combat.SilentAimFOV,
	   Flag = "SilentAimFOV",
	   Callback = function(Value)
	      Config.Combat.SilentAimFOV = Value
	   end,
	})

	CombatTab:CreateSection("Aimbot Training")

	CombatTab:CreateToggle({
	   Name = "Aimbot Enable",
	   CurrentValue = Config.Combat.Aimbot,
	   Flag = "Aimbot",
	   Callback = function(Value)
	      Config.Combat.Aimbot = Value
	   end,
	})

	---------------------------------------------------------
	-- 2. AIM SETTINGS TAB
	---------------------------------------------------------
	AimTab:CreateSection("FOV & Hitbox Configuration")

	AimTab:CreateToggle({
	   Name = "Draw Aimbot FOV Circle",
	   CurrentValue = Config.Combat.DrawAimbotFOV,
	   Flag = "DrawAimbotFOV",
	   Callback = function(Value)
	      Config.Combat.DrawAimbotFOV = Value
	   end,
	})

	AimTab:CreateSlider({
	   Name = "Aimbot FOV Radius",
	   Range = {50, 400},
	   Increment = 5,
	   Suffix = "px",
	   CurrentValue = Config.Combat.AimbotFOV,
	   Flag = "AimbotFOV",
	   Callback = function(Value)
	      Config.Combat.AimbotFOV = Value
	   end,
	})

	AimTab:CreateDropdown({
	   Name = "Target Hitbox",
	   Options = {"Head", "Torso", "HumanoidRootPart"},
	   CurrentOption = {Config.Combat.TargetPart},
	   MultipleOptions = false,
	   Flag = "TargetPart",
	   Callback = function(Option)
	      if type(Option) == "table" then
	         Config.Combat.TargetPart = Option[1]
	      else
	         Config.Combat.TargetPart = Option
	      end
	   end,
	})

	AimTab:CreateSlider({
	   Name = "Aimbot Smoothing",
	   Range = {2, 25},
	   Increment = 1,
	   Suffix = "",
	   CurrentValue = Config.Combat.Smoothing,
	   Flag = "Smoothing",
	   Callback = function(Value)
	      Config.Combat.Smoothing = Value
	   end,
	})

	---------------------------------------------------------
	-- 3. VISUALS TAB
	---------------------------------------------------------
	VisualsTab:CreateSection("ESP Suite")

	VisualsTab:CreateToggle({
	   Name = "Master ESP Switch",
	   CurrentValue = Config.Visuals.ESPEnabled,
	   Flag = "ESPEnabled",
	   Callback = function(Value)
	      Config.Visuals.ESPEnabled = Value
	   end,
	})

	VisualsTab:CreateToggle({
	   Name = "Box ESP",
	   CurrentValue = Config.Visuals.BoxESP,
	   Flag = "BoxESP",
	   Callback = function(Value)
	      Config.Visuals.BoxESP = Value
	   end,
	})

	VisualsTab:CreateToggle({
	   Name = "Tracers",
	   CurrentValue = Config.Visuals.Tracers,
	   Flag = "Tracers",
	   Callback = function(Value)
	      Config.Visuals.Tracers = Value
	   end,
	})

	---------------------------------------------------------
	-- 4. MISCELLANEOUS TAB
	---------------------------------------------------------
	MiscTab:CreateSection("Character Enhancements")

	MiscTab:CreateSlider({
	   Name = "WalkSpeed Modifier",
	   Range = {16, 120},
	   Increment = 1,
	   Suffix = "spd",
	   CurrentValue = Config.Misc.SpeedMultiplier,
	   Flag = "WalkSpeed",
	   Callback = function(Value)
	      Config.Misc.SpeedMultiplier = Value
	      local char = LocalPlayer.Character
	      if char and char:FindFirstChildOfClass("Humanoid") then
	         char:FindFirstChildOfClass("Humanoid").WalkSpeed = Value
	      end
	   end,
	})

	MiscTab:CreateSlider({
	   Name = "JumpPower Modifier",
	   Range = {50, 200},
	   Increment = 1,
	   Suffix = "jp",
	   CurrentValue = Config.Misc.JumpPower,
	   Flag = "JumpPower",
	   Callback = function(Value)
	      Config.Misc.JumpPower = Value
	      local char = LocalPlayer.Character
	      if char and char:FindFirstChildOfClass("Humanoid") then
	         char:FindFirstChildOfClass("Humanoid").JumpPower = Value
	      end
	   end,
	})

	---------------------------------------------------------
	-- 5. SETTINGS TAB
	---------------------------------------------------------
	SettingsTab:CreateSection("Interface Management")

	SettingsTab:CreateButton({
	   Name = "Destroy GUI",
	   Callback = function()
	      Rayfield:Destroy()
	   end,
	})

	---------------------------------------------------------
	-- BACKEND ENGINE & TARGETING LOGIC
	---------------------------------------------------------

	-- Character Respawn Speed & Jump Persistence
	LocalPlayer.CharacterAdded:Connect(function(char)
		local humanoid = char:WaitForChild("Humanoid", 5)
		if humanoid then
			humanoid.WalkSpeed = Config.Misc.SpeedMultiplier
			humanoid.JumpPower = Config.Misc.JumpPower
		end
	end)

	-- Drawing API Setup (Two separate circles for Silent Aim and Aimbot)
	local DrawingSupported, DrawingLib = pcall(function() return Drawing end)
	local SilentCircle = nil
	local AimbotCircle = nil

	if DrawingSupported and DrawingLib then
		pcall(function()
			-- Silent Aim Circle (Green/Cyan)
			SilentCircle = Drawing.new("Circle")
			SilentCircle.Thickness = 1.5
			SilentCircle.NumSides = 48
			SilentCircle.Filled = false
			SilentCircle.Transparency = 0.8
			SilentCircle.Color = Color3.fromRGB(0, 255, 120)
			SilentCircle.Visible = false

			-- Aimbot Circle (Purple)
			AimbotCircle = Drawing.new("Circle")
			AimbotCircle.Thickness = 1.5
			AimbotCircle.NumSides = 48
			AimbotCircle.Filled = false
			AimbotCircle.Transparency = 0.8
			AimbotCircle.Color = Color3.fromRGB(115, 87, 255)
			AimbotCircle.Visible = false
		end)
	end

	-- Helper: Get closest target within a specific radius from screen center
	local function getTargetInRadius(maxRadius)
		local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
		local closestTarget = nil
		local shortestDist = maxRadius

		for _, player in ipairs(Players:GetPlayers()) do
			if player ~= LocalPlayer and player.Character then
				local humanoid = player.Character:FindFirstChildOfClass("Humanoid")
				local targetPart = player.Character:FindFirstChild(Config.Combat.TargetPart)

				if humanoid and humanoid.Health > 0 and targetPart then
					local screenPos, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
					if onScreen then
						local dist = (Vector2.new(screenPos.X, screenPos.Y) - screenCenter).Magnitude
						if dist < shortestDist then
							shortestDist = dist
							closestTarget = targetPart
						end
					end
				end
			end
		end
		return closestTarget
	end

	-- Silent Aim Raycast Hook (Curves bullets towards targets inside Silent Aim circle)
	pcall(function()
		local oldNamecall
		oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
			local method = getnamecallmethod()
			local args = {...}

			if Config.Combat.SilentAim and self == workspace and (method == "Raycast" or method == "FindPartOnRay" or method == "FindPartOnRayWithIgnoreList") then
				local targetPart = getTargetInRadius(Config.Combat.SilentAimFOV)
				if targetPart then
					if method == "Raycast" then
						local origin = args[1]
						local direction = (targetPart.Position - origin).Unit * args[2].Magnitude
						args[2] = direction
					elseif method == "FindPartOnRay" or method == "FindPartOnRayWithIgnoreList" then
						local ray = args[1]
						local direction = (targetPart.Position - ray.Origin).Unit * ray.Direction.Magnitude
						args[1] = Ray.new(ray.Origin, direction)
					end
				end
			end

			return oldNamecall(self, unpack(args))
		end)
	end)

	-- ESP Storage Table
	local ESPObjects = {}
	local function removeESP(player)
		if ESPObjects[player] then
			if ESPObjects[player].Box then ESPObjects[player].Box:Remove() end
			if ESPObjects[player].Tracer then ESPObjects[player].Tracer:Remove() end
			ESPObjects[player] = nil
		end
	end

	-- Main Runtime Render Loop
	RunService.RenderStepped:Connect(function()
		local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

		-- 1. Update Silent Aim Circle Display
		if SilentCircle then
			SilentCircle.Visible = Config.Combat.DrawSilentFOV
			SilentCircle.Radius = Config.Combat.SilentAimFOV
			SilentCircle.Position = screenCenter
		end

		-- 2. Update Aimbot Circle Display
		if AimbotCircle then
			AimbotCircle.Visible = Config.Combat.DrawAimbotFOV
			AimbotCircle.Radius = Config.Combat.AimbotFOV
			AimbotCircle.Position = screenCenter
		end

		-- 3. Aimbot Execution
		if Config.Combat.Aimbot then
			local target = getTargetInRadius(Config.Combat.AimbotFOV)
			if target then
				local currentCFrame = Camera.CFrame
				local targetCFrame = CFrame.new(currentCFrame.Position, target.Position)
				Camera.CFrame = currentCFrame:Lerp(targetCFrame, 1 / Config.Combat.Smoothing)
			end
		end

		-- 4. ESP Rendering Loop
		if DrawingSupported and DrawingLib then
			for _, player in ipairs(Players:GetPlayers()) do
				if player ~= LocalPlayer then
					if Config.Visuals.ESPEnabled and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
						local char = player.Character
						local humanoid = char:FindFirstChildOfClass("Humanoid")
						local rootPart = char:FindFirstChild("HumanoidRootPart")

						if humanoid and humanoid.Health > 0 and rootPart then
							local rootPos, onScreen = Camera:WorldToViewportPoint(rootPart.Position)

							if not ESPObjects[player] then
								ESPObjects[player] = {
									Box = Drawing.new("Square"),
									Tracer = Drawing.new("Line")
								}
								ESPObjects[player].Box.Thickness = 1
								ESPObjects[player].Box.Filled = false
								ESPObjects[player].Box.Color = Color3.fromRGB(255, 60, 100)

								ESPObjects[player].Tracer.Thickness = 1
								ESPObjects[player].Tracer.Color = Color3.fromRGB(255, 60, 100)
							end

							local espData = ESPObjects[player]

							if onScreen then
								if Config.Visuals.BoxESP then
									local head = char:FindFirstChild("Head")
									if head then
										local headPos = Camera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0))
										local legPos = Camera:WorldToViewportPoint(rootPart.Position - Vector3.new(0, 3, 0))
										local height = math.abs(headPos.Y - legPos.Y)
										local width = height / 2

										espData.Box.Size = Vector2.new(width, height)
										espData.Box.Position = Vector2.new(rootPos.X - width / 2, headPos.Y)
										espData.Box.Visible = true
									else
										espData.Box.Visible = false
									end
								else
									espData.Box.Visible = false
								end

								if Config.Visuals.Tracers then
									espData.Tracer.From = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y)
									espData.Tracer.To = Vector2.new(rootPos.X, rootPos.Y)
									espData.Tracer.Visible = true
								else
									espData.Tracer.Visible = false
								end
							else
								espData.Box.Visible = false
								espData.Tracer.Visible = false
							end
						else
							removeESP(player)
						end
					else
						removeESP(player)
					end
				end
			end
		end
	end)

	Rayfield:LoadConfiguration()
end)

if not success then
	warn("[Mobile FPS Suite Error]: " .. tostring(err))
end
